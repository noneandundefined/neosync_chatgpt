BEGIN

        SELECT 1 FROM pg_constraint WHERE conname = 'fk_user_roles_role' AND conrelid = 'user_roles'::regclass

        IF EXISTS (
            SELECT 1 FROM pg_constraint WHERE conname = 'fk_role_code' AND conrelid = 'user_roles'::regclass
